# mycallyplus_v1/pipeline/algo/zhao2020.py
# Zhao et al. RTSS 2020: CPC model + rule-based priority assignment (EA)
#
# Paper: "DAG Scheduling and Analysis on Multiprocessor Systems: Exploitation of Parallelism and Dependency"
# Key refs:
# - Algorithm 1: CPC(G, λ*): CPC model construction
# - Rule 1/2/3*: priority rules + nested CPC recursion

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, List, Set, Tuple, Optional, Iterable


NodeId = str


@dataclass(frozen=True)
class Edge:
    src: NodeId
    dst: NodeId


@dataclass
class Dag:
    nodes: List[NodeId]
    edges: List[Edge]

    # Optional helpers (you can replace with networkx if your project already uses it)
    def succ(self) -> Dict[NodeId, List[NodeId]]:
        out: Dict[NodeId, List[NodeId]] = {u: [] for u in self.nodes}
        for e in self.edges:
            out[e.src].append(e.dst)
        return out

    def pred(self) -> Dict[NodeId, List[NodeId]]:
        inc: Dict[NodeId, List[NodeId]] = {u: [] for u in self.nodes}
        for e in self.edges:
            inc[e.dst].append(e.src)
        return inc


@dataclass
class Provider:
    """A provider θ*_i is a consecutive sub-path on the critical path."""
    idx: int
    nodes: List[NodeId]  # in topological order along λ*
    # consumer group F(θ*_i) and early-interference set G(θ*_i)
    F: Set[NodeId]
    G: Set[NodeId]


@dataclass
class CPCModel:
    """CPC model: Θ* providers and Θ consumers (non-critical nodes)."""
    providers: List[Provider]
    critical_path: List[NodeId]
    critical_set: Set[NodeId]
    consumers_all: Set[NodeId]  # Θ = V \ λ*


@dataclass
class Zhao2020Result:
    priorities: Dict[NodeId, int]
    meta: Dict


# -----------------------------
# Public entry
# -----------------------------
def schedule_zhao2020(
    dag: Dag,
    wcet: Dict[NodeId, float],
    prio_max: int = 80,
    *,
    tie_break: str = "node_id",   # stable reproducibility
) -> Zhao2020Result:
    """
    Return fixed priorities for nodes using Zhao2020 (CPC + EA rule-based assignment).

    Inputs:
      - dag: DAG (single source/sink assumed by the paper, but you can relax with virtual src/sink)
      - wcet: node weight (execution time); use your segment timing
      - prio_max: highest integer priority to assign, then decrement

    Output:
      - priorities: node_id -> int priority (higher is more important)
      - meta: debug info (critical path, providers, F/G sizes, local-path lengths, etc.)
    """
    _validate_dag(dag)

    # 1) Find critical path λ* (longest complete path by sum of wcet)
    critical_path = _critical_path_longest_by_wcet(dag, wcet, tie_break=tie_break)

    # 2) Build CPC model Θ*, and per-provider F(θ*_i), G(θ*_i) using Algorithm 1
    cpc = _build_cpc_model(dag, critical_path)

    # 3) EA priority assignment (Algorithm 2) using Rules 1, 2, 3* + nested CPC recursion
    priorities = _ea_priority_assignment(
        dag=dag,
        wcet=wcet,
        cpc=cpc,
        pmax=prio_max,
        tie_break=tie_break,
    )

    meta = {
        "algo": "zhao2020",
        "prio_max": prio_max,
        "critical_path": critical_path,
        "providers": [
            {
                "idx": pr.idx,
                "nodes": pr.nodes,
                "F": sorted(pr.F),
                "G": sorted(pr.G),
            }
            for pr in cpc.providers
        ],
    }
    return Zhao2020Result(priorities=priorities, meta=meta)


# -----------------------------
# Step 1: Critical path
# -----------------------------
def _critical_path_longest_by_wcet(
    dag: Dag,
    wcet: Dict[NodeId, float],
    *,
    tie_break: str,
) -> List[NodeId]:
    """
    Compute critical path λ* as the longest path (sum of wcet).
    Implementation: DP on reverse topological order + reconstruct path.

    Note: The paper defines critical path as longest complete path from source to sink. :contentReference[oaicite:3]{index=3}
    """
    topo = _topological_sort(dag)
    succ = dag.succ()

    # dp[u] = wcet[u] + max(dp[v]) for v in succ(u); sink dp = wcet[sink]
    dp: Dict[NodeId, float] = {}
    next_choice: Dict[NodeId, Optional[NodeId]] = {}

    for u in reversed(topo):
        best_v = None
        best = float("-inf")
        for v in succ[u]:
            val = dp.get(v, float("-inf"))
            if val > best or (val == best and _tie(v, best_v, tie_break)):
                best = val
                best_v = v
        dp[u] = float(wcet.get(u, 0.0)) + (best if best_v is not None else 0.0)
        next_choice[u] = best_v

    # pick a source (or virtual source). If multiple, pick the one with max dp.
    sources = [u for u in dag.nodes if len(_pred_map(dag)[u]) == 0]
    if not sources:
        raise ValueError("DAG has no source")
    src = max(sources, key=lambda u: (dp[u], u if tie_break == "node_id" else 0))

    # reconstruct
    path = [src]
    cur = src
    while next_choice[cur] is not None:
        cur = next_choice[cur]  # type: ignore
        path.append(cur)
    return path


# -----------------------------
# Step 2: CPC model (Algorithm 1)
# -----------------------------
def _build_cpc_model(dag: Dag, critical_path: List[NodeId]) -> CPCModel:
    """
    Implements Algorithm 1 CPC(G, λ*). :contentReference[oaicite:4]{index=4}

    - Step 1: Split λ* into consecutive providers θ*_i based on dependency condition:
      while pre(v_{j+1}) == {v_j} keep extending provider. :contentReference[oaicite:5]{index=5}
    - Step 2: For each provider, compute:
        F(θ*_i) = anc(θ*_{i+1}) ∩ V¬  (nodes that can delay next provider)
        G(θ*_i) = ⋃_{v in F(θ*_i)} (C(v) ∩ V¬)  (early-released consumers from later groups)
      and remove assigned F from V¬ progressively. :contentReference[oaicite:6]{index=6}
    """
    pred = _pred_map(dag)
    critical_set = set(critical_path)
    consumers = set(dag.nodes) - critical_set

    # Step 1: providers (split critical path)
    providers_nodes: List[List[NodeId]] = []
    i = 0
    while i < len(critical_path):
        cur = [critical_path[i]]
        j = i
        while j + 1 < len(critical_path):
            vj = critical_path[j]
            vnext = critical_path[j + 1]
            # pre(v_{j+1}) == {v_j}
            if set(pred[vnext]) == {vj}:
                cur.append(vnext)
                j += 1
            else:
                break
        providers_nodes.append(cur)
        i = j + 1

    # Step 2: compute F/G per provider
    providers: List[Provider] = []
    V_not = set(consumers)  # V¬ changes as Algorithm 1 line 14

    # precompute ancestors, and concurrency sets C(v)
    anc_map = _ancestors_map(dag)
    conc_map = _concurrent_map(dag)  # C(v) = nodes not in anc(v)∪des(v)∪{v} (paper def)
    # Note: paper uses C(vj) and intersects with V¬. :contentReference[oaicite:7]{index=7}

    for idx, prov_nodes in enumerate(providers_nodes, start=1):
        # θ*_{i+1} head: in paper F(θ*_i)=anc(θ*_{i+1})∩V¬ (line 12).
        # We'll treat "next provider head node" as the first node of next provider.
        if idx < len(providers_nodes):
            next_head = providers_nodes[idx][0]
            F = set(anc_map[next_head]) & V_not
        else:
            F = set()

        # G(θ*_i) = union_{v in F} (C(v) ∩ V¬)
        G = set()
        for v in F:
            G |= (conc_map[v] & V_not)

        # remove F from V¬ (Algorithm 1 line 14)
        V_not -= F

        providers.append(Provider(idx=idx, nodes=prov_nodes, F=F, G=G))

    return CPCModel(
        providers=providers,
        critical_path=critical_path,
        critical_set=critical_set,
        consumers_all=consumers,
    )


# -----------------------------
# Step 3: EA priority assignment (Algorithm 2)
# -----------------------------
def _ea_priority_assignment(
    *,
    dag: Dag,
    wcet: Dict[NodeId, float],
    cpc: CPCModel,
    pmax: int,
    tie_break: str,
) -> Dict[NodeId, int]:
    """
    Rule-based fixed-priority assignment (EA).

    Must enforce:
    - Rule 1: all critical nodes (Θ*) higher than any consumer (Θ). :contentReference[oaicite:8]{index=8}
    - Rule 2: for i<l, all consumers in F(θ*_i) higher than all consumers in F(θ*_l). :contentReference[oaicite:9]{index=9}
    - Rule 3*: inside each F(θ*_i), assign higher priority to nodes on longer *local paths* lj(F(θ*_i)),
      dependency-aware; and recursively build nested CPC models until local paths are independent. :contentReference[oaicite:10]{index=10}
    """
    p = pmax
    prio: Dict[NodeId, int] = {u: -1 for u in dag.nodes}

    # ---------- Rule 1: critical nodes highest ----------
    # NOTE: paper says "∀ vj ∈ Θ*, pj > pk for any vk ∈ Θ". :contentReference[oaicite:11]{index=11}
    # We also need an internal order among critical nodes. Use their order along λ* for determinism.
    for u in cpc.critical_path:
        prio[u] = p
        p -= 1

    # ---------- Rule 2 + Rule 3*: consumers grouped by provider ----------
    # Paper: iterate θ*_i in topological order (provider order). :contentReference[oaicite:12]{index=12}
    for prov in cpc.providers:
        if not prov.F:
            continue

        # Assign all nodes in this F group before moving to next provider (Rule 2).
        # Within the group, order by dependency-aware longest local path (Rule 3*),
        # but if the group itself is a complex DAG, recursively apply nested CPC. :contentReference[oaicite:13]{index=13}
        ordered_nodes = _order_consumer_group_dependency_aware(
            dag=dag,
            wcet=wcet,
            group=set(prov.F),
            tie_break=tie_break,
        )

        for u in ordered_nodes:
            if prio[u] != -1:
                continue  # defensive: should not happen if groups are disjoint after Algorithm 1 removal
            prio[u] = p
            p -= 1

    # Any leftover non-critical nodes (not in any F because of model assumptions) go last.
    leftovers = [u for u in dag.nodes if prio[u] == -1]
    leftovers.sort(key=lambda u: u if tie_break == "node_id" else u)
    for u in leftovers:
        prio[u] = p
        p -= 1

    return prio


def _order_consumer_group_dependency_aware(
    *,
    dag: Dag,
    wcet: Dict[NodeId, float],
    group: Set[NodeId],
    tie_break: str,
) -> List[NodeId]:
    """
    Implements Rule 3* ordering inside F(θ*_i):

    - For nodes vj, vk in the same F group:
        if lj(F) > lk(F) then pj > pk. :contentReference[oaicite:14]{index=14}
      where lj(F) is the length of the longest local path in the subgraph induced by F that includes vj,
      computed by traversing anc(vj) ∪ des(vj) within F. :contentReference[oaicite:15]{index=15}

    - BUT: simply sorting by lj(F) is not sufficient when F forms a smaller DAG; need nested CPC recursion
      until local paths in the group are fully independent. :contentReference[oaicite:16]{index=16}

    Practical engineering approach:
      1) Build induced subgraph G_F.
      2) If G_F decomposes into multiple independent chains/paths, apply Rule 3* directly by lj.
      3) Else recursively:
         - compute critical path inside G_F
         - build CPC(G_F, λ*_F)
         - apply Rule 1/2 within that nested model
         - only apply Rule 3* to independent paths as the paper notes. :contentReference[oaicite:17]{index=17}
    """
    sub = _induced_subdag(dag, group)

    # Base case: if subgraph has no edges, they are fully independent
    if not sub.edges:
        nodes = sorted(sub.nodes, key=lambda u: u if tie_break == "node_id" else u)
        return nodes

    # Heuristic base case: if the subgraph is already a set of independent chains/paths,
    # we can compute local longest-path-through-node lj and sort.
    if _looks_like_independent_paths(sub):
        lj = _local_longest_path_through_each_node(sub, wcet)
        nodes = list(sub.nodes)
        nodes.sort(key=lambda u: (-lj[u], u if tie_break == "node_id" else u))
        return nodes

    # Recursive (nested CPC):
    # - Critical path inside this consumer group
    crit = _critical_path_longest_by_wcet(sub, wcet, tie_break=tie_break)
    nested = _build_cpc_model(sub, crit)

    # Reuse EA assignment inside this subgraph, then return nodes ordered by assigned priority desc
    nested_prio = _ea_priority_assignment(
        dag=sub,
        wcet=wcet,
        cpc=nested,
        pmax=10_000_000,  # large local pmax; we only need relative order
        tie_break=tie_break,
    )
    nodes = list(sub.nodes)
    nodes.sort(key=lambda u: (-nested_prio[u], u if tie_break == "node_id" else u))
    return nodes


# -----------------------------
# Graph utilities (replace with your existing helpers/networkx)
# -----------------------------
def _validate_dag(dag: Dag) -> None:
    topo = _topological_sort(dag)
    if len(topo) != len(dag.nodes):
        raise ValueError("DAG contains a cycle or disconnected nodes not handled")


def _topological_sort(dag: Dag) -> List[NodeId]:
    pred = _pred_map(dag)
    succ = dag.succ()
    indeg = {u: len(pred[u]) for u in dag.nodes}
    q = [u for u in dag.nodes if indeg[u] == 0]
    q.sort()
    out: List[NodeId] = []
    while q:
        u = q.pop(0)
        out.append(u)
        for v in succ[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
                q.sort()
    return out


def _pred_map(dag: Dag) -> Dict[NodeId, List[NodeId]]:
    return dag.pred()


def _ancestors_map(dag: Dag) -> Dict[NodeId, Set[NodeId]]:
    # BFS/DFS from each node over preds. (You can memoize or use transitive closure via networkx.)
    pred = _pred_map(dag)
    anc: Dict[NodeId, Set[NodeId]] = {u: set() for u in dag.nodes}
    for u in dag.nodes:
        stack = list(pred[u])
        seen = set()
        while stack:
            x = stack.pop()
            if x in seen:
                continue
            seen.add(x)
            anc[u].add(x)
            stack.extend(pred[x])
    return anc


def _descendants_map(dag: Dag) -> Dict[NodeId, Set[NodeId]]:
    succ = dag.succ()
    des: Dict[NodeId, Set[NodeId]] = {u: set() for u in dag.nodes}
    for u in dag.nodes:
        stack = list(succ[u])
        seen = set()
        while stack:
            x = stack.pop()
            if x in seen:
                continue
            seen.add(x)
            des[u].add(x)
            stack.extend(succ[x])
    return des


def _concurrent_map(dag: Dag) -> Dict[NodeId, Set[NodeId]]:
    # C(v) = V \ (anc(v) ∪ des(v) ∪ {v})
    anc = _ancestors_map(dag)
    des = _descendants_map(dag)
    V = set(dag.nodes)
    out: Dict[NodeId, Set[NodeId]] = {}
    for v in dag.nodes:
        out[v] = V - anc[v] - des[v] - {v}
    return out


def _induced_subdag(dag: Dag, nodes: Set[NodeId]) -> Dag:
    edges = [e for e in dag.edges if e.src in nodes and e.dst in nodes]
    return Dag(nodes=sorted(nodes), edges=edges)


def _looks_like_independent_paths(sub: Dag) -> bool:
    # Simple structural check (you can refine):
    # if every node has outdegree<=1 and indegree<=1 then it's a union of chains.
    pred = sub.pred()
    succ = sub.succ()
    for u in sub.nodes:
        if len(pred[u]) > 1 or len(succ[u]) > 1:
            return False
    return True


def _local_longest_path_through_each_node(sub: Dag, wcet: Dict[NodeId, float]) -> Dict[NodeId, float]:
    """
    Compute lj(F): length of longest local path in subgraph that includes vj.
    Engineering implementation:
      lj(v) = best_path_ending_at_v + best_path_starting_from_v - wcet[v]
    where best_path_ending_at_v is max sum along preds within subgraph,
    and best_path_starting_from_v is max sum along succs within subgraph.
    """
    topo = _topological_sort(sub)
    pred = sub.pred()
    succ = sub.succ()

    best_end: Dict[NodeId, float] = {}
    for v in topo:
        best = 0.0
        for p in pred[v]:
            best = max(best, best_end[p])
        best_end[v] = float(wcet.get(v, 0.0)) + best

    best_start: Dict[NodeId, float] = {}
    for v in reversed(topo):
        best = 0.0
        for s in succ[v]:
            best = max(best, best_start[s])
        best_start[v] = float(wcet.get(v, 0.0)) + best

    lj: Dict[NodeId, float] = {}
    for v in sub.nodes:
        lj[v] = best_end[v] + best_start[v] - float(wcet.get(v, 0.0))
    return lj


def _tie(a: Optional[NodeId], b: Optional[NodeId], tie_break: str) -> bool:
    if b is None:
        return True
    if a is None:
        return False
    if tie_break == "node_id":
        return a < b
    return False

## 你接下来需要 Codex 补齐/确认的 3 个点（非常关键）

1. **你们 DAG 的“source/sink”**
   论文默认单 source/sink 。
   如果你们有多个入口/出口，建议在构图阶段加 **虚拟 source/sink** 再跑 zhao2020。

2. **Rule 3* 的递归终止条件**
   论文强调：只有当 consumer group 的 local paths “fully independent” 才直接用 Rule 3* 排序，否则要 nested CPC 递归 。
   我在骨架里给了一个 `_looks_like_independent_paths()` 的工程判断，你可以替换为更严格的“路径分解”。

3. **与你们 pipeline 的输出契约**
   你们现在 `schedule.json` 里至少要有 `priorities`，最好把 `meta.providers/F/G/critical_path` 都塞进去，方便 GUI/调试（你前面提到你们 HEFT 会写 meta.rank_u，这里同理）。
